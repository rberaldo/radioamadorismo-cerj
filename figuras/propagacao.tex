\documentclass{cerj-figura}
% \propagacao{n} desenha o passo n; todas as páginas têm o mesmo tamanho
\newcommand{\propagacao}[1]{%
  \begin{tikzpicture}[scale=.9,
      rotulo/.style={font=\scriptsize, text=cerjInk},
      icone/.style={text=cerjNavy, anchor=south, inner sep=1pt}]
    \useasboundingbox (-6.9,-.45) rectangle (6.9,5.2);
  
    % Ionosfera
    \fill[cerjNavy10] (-6.9,4.4) rectangle (6.9,4.9);
    \node[rotulo, text=cerjNavy60, anchor=east] at (6.8,4.65) {ionosfera};
  
    % Terra (curvatura exagerada) e morro
    \fill[cerjGreen!25] plot[domain=-6.9:6.9, samples=40] (\x, {.35*(1-(\x/6.9)^2)}) -- (6.9,-.45) -- (-6.9,-.45) -- cycle;
    \fill[cerjGreenDark!55] (-1.6,.3) -- (-.4,1.5) -- (.3,1.25) -- (1.2,2.4) -- (2.2,1.3) -- (3.1,.27) -- cycle;
  
    % Estações e montanhistas
    \node[icone] (base)  at (-5.8,.1)  {\faCampground};
    \node[rotulo, below=0pt of base] {base};
    \node[icone] (longe) at ( 6.0,.08) {\faHome};
    \node[rotulo, below=0pt of longe, xshift=-3mm] {centenas de km};
    \node[icone] (m1) at (-2.3,.31) {\faHiking};
    \node[icone] (m2) at ( 3.8,.24) {\faHiking};
  
    % 1. HF reflete na ionosfera
    \draw[cerjBlue, thick, -{Stealth}] (base.north) -- (0,4.4) -- (longe.north);
    \node[rotulo, text=cerjBlue, font=\scriptsize\bfseries, anchor=south] at (0,4.4) {HF};
  
    % 2. VHF/UHF atravessa a ionosfera; o morro bloqueia a visada
    \ifnum#1>1
      \draw[cerjGreenDark, thick, -{Stealth}] (base.north) -- (-4.4,5.2);
      \node[rotulo, text=cerjGreenDark, font=\scriptsize\bfseries, anchor=west] at (-4.35,5.05) {VHF/UHF};
      \draw[cerjRed, thick, dashed] (m1.north) -- (m2.north);
      \node[text=cerjRed, fill=cerjPaper, inner sep=1pt, font=\small] at (.75,.75) {\faTimes};
    \fi
  
    % 3. Quem está no cume enxerga os dois lados
    \ifnum#1>2
      \node[icone] (cume) at (1.2,2.28) {\faHiking};
      \draw[cerjGreenDark, thick] (base.north) -- (cume.west);
      \draw[cerjGreenDark, thick] (m1.north) -- (cume.west);
      \draw[cerjGreenDark, thick] (cume.east) -- (m2.north);
    \fi
  \end{tikzpicture}
  }
