\section{O rádio na montanha}

\begin{frame}{Potencialidades e limites do rádio}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \textbf{\textcolor{cerjGreenDark}{O rádio faz}}
      \begin{certos}
      \item Funciona \alert<1>{sem infraestrutura}: dois rádios bastam
      \item Fala com \alert<2>{todo o grupo de uma vez}, na caminhada e entre cordadas; \alert<3>{não há privacidade}.
      \item Contato com a base e o mundo, em linha de visada ou via repetidoras
      \item É imediato: basta apertar e falar
      \end{certos}
    \column{0.48\textwidth}
      \textbf{\textcolor{cerjRed}{O rádio não faz}}
      \begin{errados}
      \item Não garante que \alert<4>{alguém esteja ouvindo}
      \item Não vence o relevo
      \item Não resolve áreas muito remotas (ex.: PN das Sempre-Vivas)
      \item Não substitui o \alert<5>{comunicador satelital} nem as chamadas de \alert<6>{emergência do celular}
      \item Não funciona sem \alert<7>{preparo} ou licença
      \end{errados}
  \end{columns}
\end{frame}

\begin{frame}{Rádio, celular ou comunicador satelital?}
  \centering
  \small
  \renewcommand{\arraystretch}{1.3}
  \setlength{\tabcolsep}{4pt}

  \normalsize
  \pause
  O rádio é a melhor ferramenta para \alert<5->{coordenar o grupo}. Para pedir socorro, ele \alert<6>{soma}, mas nunca substitui.

\begin{frame}{Algumas dicas antes de sair de casa}
  \small
  \begin{dica}
    Transporte o rádio \alert{sem a bateria encaixada}: o knob gira na mochila, o rádio liga e
    chega descarregado. Já aconteceu.
  \end{dica}

  \pause
  \begin{dica}
    Fique atento ao \alert{squelch} (SQL): silencia o chiado quando ninguém fala; alto demais, corta sinais fracos. O subtom (CTCSS) \alert{não dá privacidade}: só faz o squelch abrir quando o sinal traz aquele tom.
  \end{dica}

  \pause
  \begin{regulamento}
    Rádios como o UV-5R transmitem nas faixas de radioamador: usá-los exige \alert{COER e licença} com indicativo. Eles não são rádios de uso geral (RUG).
  \end{regulamento}
\end{frame}

\begin{frame}{Comunicação entre o grupo ou a cordada}
  \textbf{Simplex}: comunicação direta, na mesma frequência, sem repetidora. Só funciona se houver \alert{linha de visada} entre os dois rádios.

  \medskip
  \begin{itemize}
  \item Os rádios vão \alert{nas pontas}: com quem abre e com quem cerra a fila
  \item Se o grupo se dividir, cada parte leva um rádio
  \item Todos na \alert{na mesma frequência}, combinada antes de sair e gravado na memória
  \end{itemize}

  \medskip
  \begin{atencao}
    Dois rádios lado a lado são inúteis: quem está junto conversa sem rádio.
  \end{atencao}
\end{frame}

\begin{frame}{Comunicação com o mundo}
  \textbf{Frequência Nacional de Chamada (FNC)}: \alert<1>{\freq{146,520}, em simplex}. Chame, estabeleça contato e mude de frequência; fique sempre ouvindo, para casos de emergência.

  \pause
  \medskip
  \textbf{Repetidora}: fica no alto de um morro, recebe na \alert{entrada} e retransmite na
  \alert{saída}. Assim, quem não se enxerga consegue conversar.

  \smallskip
  \centering
  \figura{repetidora}
\end{frame}

% Protocolo de Áreas Remotas, com base em Salvatti (2024, seção 8.3)
\begin{frame}{Protocolo de operação em áreas remotas}
  \begin{atencao}
   \alert{Conheça o seu equipamento} antes de precisar dele!
  \end{atencao}

  \pause
  Para poupar bateria numa emergência, ligue o rádio só quando há mais chance de alguém ouvir
  \citep{salvatti2024}:

  \medskip
  \begin{enumerate}
  \item Se puder, suba ao ponto mais alto, numa clareira ou com o horizonte desobstruído
  \item Sintonize a FNC, \freq{146,520}, ou o canal 9 da RUG, \freq{462,6625}; melhor ainda se houver uma repetidora ao alcance
  \item Ligue o rádio nas \alert{horas cheias}, escute por pelo menos 15 minutos e desligue
  \end{enumerate}

  \medskip
  \comentario{Os 15 minutos são o pior caso: com bateria extra ou power bank, dá para escutar por dias.}
\end{frame}

\begin{frame}{Protocolo de áreas remotas: pedido de socorro}
  \begin{enumerate}
  \setcounter{enumi}{4}
  \item Para pedir socorro, diga:
  \end{enumerate}

  \begin{alertblock}{Pedido de socorro}
    \centering
    ``\textbf{Preciso de ajuda}, este é \textit{\textless identificação\textgreater}, minha
    localização é \textit{\textless geolocalização\textgreater}.''
  \end{alertblock}

  \begin{enumerate}
  \setcounter{enumi}{5}
  \item Aguarde a resposta
  \item Para interromper uma conversa, diga ``\alert{break}'' três vezes
  \end{enumerate}

  \begin{atencao}
    Jamais diga ``break'' fora de uma emergência
    (ITU, 2010, p.~23, apud \citealp{salvatti2024}).
  \end{atencao}
\end{frame}

\begin{frame}{Caixas de destaque}
  \begin{dica}
    Leve uma antena telescópica de meia onda: dobra o alcance do rádio de mão.
  \end{dica}
  \begin{atencao}
    Não transmita com a antena desconectada, o estágio final do rádio queima.
  \end{atencao}
  \begin{regulamento}
    Operar sem licença é infração à Lei Geral de Telecomunicações.
  \end{regulamento}
\end{frame}

\begin{frame}{Como pedir ajuda pelo rádio}
  \begin{enumerate}
    \item Respire, escute a frequência por alguns segundos
    \item Diga seu indicativo e “\alert{emergência}” três vezes
    \item Informe \alert{onde}, \alert{quem} e \alert{o quê}
    \item Fique na escuta, não desligue o rádio
  \end{enumerate}
\end{frame}
